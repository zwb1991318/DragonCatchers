# M0 + M1：工程搭建与核心循环 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 搭好 iOS 工程（M0），并做出第一关完整可玩的核心循环（M1）：拉弓射箭 → 龙分波来袭并远程喷火 → 龙落河漂流 → 船自动打捞 → 收益结算 → 胜负判定 → 再来一次。

**Architecture:** 所有游戏规则写在本地 Swift 包 `Packages/DragonCore` 里，用纯 Swift 值类型实现，不依赖 SpriteKit，可以用 `swift test` 在命令行直接测试。App 端（`App/` 和 `Game/`）只做两件事：每帧调用 `Battle.step(dt:)`，再把状态画成 SpriteKit 节点；同时把触摸转发给 `Battle`。数值和关卡放在包内的 JSON 文件里。

**Tech Stack:** Swift 6.4、SpriteKit、SwiftUI（`SpriteView`）、Swift Testing、XcodeGen 2.46、Xcode 27 / iOS 27 SDK，最低支持 iOS 17，模拟器用 iPhone 15。

**设计依据:** `docs/GameDesign.md` v0.3

## 和设计文档的对应关系，以及 M1 的临时决定

- 设计文档 6.2 写的是 `Model/` 目录，本计划改成本地 Swift 包 `Packages/DragonCore`，目的是让纯逻辑可以脱离模拟器单独测试。Task 12 会同步更新设计文档。
- 箭的飞行轨迹不用 SpriteKit 物理引擎，改用解析公式计算，这样预瞄虚线和实际弹道完全一致，也能写单元测试。
- 以下几项设计文档里留待实现时再定，M1 先按下面的方式处理，并在 Task 12 写回文档：
  - 打捞和卸货都是瞬间完成，不耗时。
  - 船没装满时，返航途中如果出现能追上的龙尸，会掉头去捞（也就是"顺路打捞"）；装满了就直接返航。
  - 船没有打捞目标、船上也没货时，自动回码头待命。
- M1 只有一种龙（`normal`），龙沿直线飞行，没有暂停按钮。这些分别在 M3、M2 里做。

## 坐标约定

和 SpriteKit 一致：原点在左下角，y 轴向上。设计分辨率为 960×440，场景缩放模式为 `.aspectFill`。

/// 二维向量。坐标系和 SpriteKit 一致：原点在左下角，y 轴向上。
public struct Vec2: Codable, Equatable, Sendable {
    public var x: Double
    public var y: Double

    public init(_ x: Double, _ y: Double) {
        self.x = x
        self.y = y
    }

    public static let zero = Vec2(0, 0)

    public var length: Double { (x * x + y * y).squareRoot() }

    public func distance(to other: Vec2) -> Double { (self - other).length }

    public static func + (a: Vec2, b: Vec2) -> Vec2 { Vec2(a.x + b.x, a.y + b.y) }
    public static func - (a: Vec2, b: Vec2) -> Vec2 { Vec2(a.x - b.x, a.y - b.y) }
    public static func * (v: Vec2, s: Double) -> Vec2 { Vec2(v.x * s, v.y * s) }
}
```

#!/usr/bin/env bash
# 生成工程、编译，并在 iPhone 模拟器上启动游戏。
# 用法：scripts/run-sim.sh            （默认 iPhone 15）
#       DEVICE="iPhone 15 Pro" scripts/run-sim.sh
set -euo pipefail
cd "$(dirname "$0")/.."

DEVICE="${DEVICE:-iPhone 15}"
BUNDLE_ID="com.zwb1991318.DragonCatchers"

# 同名模拟器可能有多个系统版本，选系统版本最新的那个
UDID="$(xcrun simctl list devices available -j | python3 -c '
import json, sys
name = sys.argv[1]
devices = json.load(sys.stdin)["devices"]
found = sorted((rt, d["udid"]) for rt, ds in devices.items() if "iOS" in rt for d in ds if d["name"] == name)
print(found[-1][1] if found else "")
' "$DEVICE")"
if [ -z "$UDID" ]; then
  echo "找不到模拟器：$DEVICE" >&2
  exit 1
fi

xcodegen generate --quiet
xcodebuild -project DragonCatchers.xcodeproj -scheme DragonCatchers \
  -destination "id=$UDID" -derivedDataPath build \
  CODE_SIGNING_ALLOWED=NO -quiet build

APP="build/Build/Products/Debug-iphonesimulator/DragonCatchers.app"
xcrun simctl boot "$UDID" 2>/dev/null || true
open -a Simulator
xcrun simctl install "$UDID" "$APP"
xcrun simctl launch "$UDID" "$BUNDLE_ID"
echo "已在 $DEVICE（$UDID）上启动"
```

```swift
/// 可复现的随机数（SplitMix64）。同一个种子总是得到同样的序列，方便测试和复现问题。
public struct SeededRandom: RandomNumberGenerator, Sendable {
    private var state: UInt64

    public init(seed: UInt64) {
        state = seed
    }

    public mutating func next() -> UInt64 {
        state &+= 0x9E37_79B9_7F4A_7C15
        var z = state
        z = (z ^ (z >> 30)) &* 0xBF58_476D_1CE4_E5B9
        z = (z ^ (z >> 27)) &* 0x94D0_49BB_1331_11EB
        return z ^ (z >> 31)
    }
}
```

/// 全局数值配置，对应 Data/balance.json。
public struct BalanceConfig: Codable, Equatable, Sendable {
    public var world: WorldConfig
    public var bow: BowConfig
    public var wall: WallConfig
    public var river: RiverConfig
    public var boat: BoatConfig
    public var economy: EconomyConfig
    public var dragons: [String: DragonKind]
    /// 胜利后留给船继续打捞的收尾时间（秒）
    public var victoryGraceTime: Double
}

public struct WorldConfig: Codable, Equatable, Sendable {
    public var width: Double
    public var height: Double
    /// 重力加速度（正数，方向向下）
    public var gravity: Double
    /// 龙出现的位置：屏幕右侧外的 x 坐标，以及 y 的随机范围
    public var spawnX: Double
    public var spawnMinY: Double
    public var spawnMaxY: Double
}

public struct BowConfig: Codable, Equatable, Sendable {
    /// 弓箭手位置，也是箭的发射点
    public var anchor: Vec2
    /// 可起手区域半径，约为弓箭手图形的 2 倍
    public var grabRadius: Double
    /// 取消圈半径：手指离按下点不超过这个距离时松手，视为取消
    public var cancelRadius: Double
    /// 最大拉弓距离，超过后力度不再增加
    public var maxPull: Double
    /// 每拉 1 点距离对应的箭初速度
    public var launchSpeedPerPoint: Double
    public var reloadTime: Double
    public var damage: Double
}

public struct WallConfig: Codable, Equatable, Sendable {
    /// 城墙正面（朝向龙的一侧）的 x 坐标
    public var frontX: Double
    public var topY: Double
    public var maxHP: Double
}

public struct RiverConfig: Codable, Equatable, Sendable {
    /// 水面高度：箭低于它就算落水
    public var surfaceY: Double

/// 箭的抛物线。实际飞行和预瞄虚线用同一套公式，保证两者完全一致。
public enum Trajectory {
    public static func position(origin: Vec2, velocity: Vec2, gravity: Double, time t: Double) -> Vec2 {
        Vec2(origin.x + velocity.x * t, origin.y + velocity.y * t - 0.5 * gravity * t * t)
    }

    public static func velocity(launch: Vec2, gravity: Double, time t: Double) -> Vec2 {
        Vec2(launch.x, launch.y - gravity * t)
    }

    /// 预瞄虚线上的点：从发射后 step 秒开始，每隔 step 秒取一个点
    public static func previewPoints(origin: Vec2, velocity: Vec2, gravity: Double, step: Double, count: Int) -> [Vec2] {
        guard count > 0 else { return [] }
        return (1...count).map { position(origin: origin, velocity: velocity, gravity: gravity, time: Double($0) * step) }
    }
}

public enum ShotResult: Equatable, Sendable {
    case fired(velocity: Vec2)
    case cancelled
}

/// 拉弓输入的状态机：只负责把"按下—拖动—松手"变成一次发射或取消，不管箭怎么飞。
public struct Bow: Sendable {
    public let config: BowConfig
    public private(set) var reloadRemaining: Double = 0
    /// 按下的位置（起点）；为 nil 表示当前没在拉弓
    public private(set) var start: Vec2?
    public private(set) var current: Vec2?

    public init(config: BowConfig) {
        self.config = config
    }

    public var isAiming: Bool { start != nil }
    public var isReady: Bool { reloadRemaining <= 0 }

    public mutating func update(dt: Double) {
        reloadRemaining = max(0, reloadRemaining - dt)
    }

    /// 只有按在弓箭手附近才开始拉弓
    @discardableResult
    public mutating func begin(at point: Vec2) -> Bool {
        guard point.distance(to: config.anchor) <= config.grabRadius else { return false }
        start = point
        current = point
        return true
    }

    public mutating func move(to point: Vec2) {
        guard isAiming else { return }
        current = point
    }

    /// 当前拉弓对应的箭初速度：方向和拖动方向相反，大小随拖动距离增加，超过 maxPull 后不再增加。
    /// 手指还在取消圈内时返回 nil。
    public var aimVelocity: Vec2? {
        guard let start, let current else { return nil }
        let pull = start - current
        let distance = pull.length
        guard distance > config.cancelRadius else { return nil }
        let clamped = min(distance, config.maxPull)
        return pull * (clamped / distance * config.launchSpeedPerPoint)
    }

    /// 松手。装填没完成，或者手指在取消圈内，都视为取消。
    public mutating func release() -> ShotResult {
        defer { cancel() }
        guard isReady, let velocity = aimVelocity else { return .cancelled }
        reloadRemaining = config.reloadTime
        return .fired(velocity: velocity)
    }

    public mutating func cancel() {
        start = nil
        current = nil
    }
}

/// 按关卡配置决定什么时候出哪条龙。只管出龙，不管龙的死活；调用方每帧告诉它场上还有几条活龙。
public struct WaveSpawner: Sendable {
    private enum State: Equatable, Sendable {
        /// 等待本波开始
        case delay(remaining: Double)
        /// 正在出本波的龙，timer 是距离下一条龙出场的时间
        case spawning(timer: Double)
        /// 本波出完了，等场上的龙全部死亡
        case waitingForClear
        case finished
    }

    private struct PendingSpawn: Equatable, Sendable {
        var kind: String
        /// 距离上一条龙出场的间隔
        var wait: Double
    }

    private let waves: [WaveConfig]
    /// 当前波次（从 0 开始）
    public private(set) var waveIndex = 0
    private var state: State
    private var queue: [PendingSpawn] = []

    public init(level: LevelConfig) {
        waves = level.waves
        state = waves.isEmpty ? .finished : .delay(remaining: waves[0].delay)
    }

    public var waveCount: Int { waves.count }

    /// 所有波次的龙都已经出完
    public var isFinished: Bool { state == .finished }

    /// 推进时间，返回这一帧要生成的龙的种类
    public mutating func step(dt: Double, aliveDragons: Int) -> [String] {
        switch state {
        case .finished:
            return []
        case .waitingForClear:
            guard aliveDragons == 0 else { return [] }
            waveIndex += 1
            state = .delay(remaining: waves[waveIndex].delay)
            return []
        case .delay(let remaining):
            let left = remaining - dt
            guard left <= 0 else {
                state = .delay(remaining: left)
                return []
            }
            queue = Self.schedule(for: waves[waveIndex])
            return advanceSpawning(timer: 0)
        case .spawning(let timer):
            return advanceSpawning(timer: timer - dt)
        }
    }

    private mutating func advanceSpawning(timer: Double) -> [String] {
        var timer = timer
        var spawned: [String] = []
        while timer <= 0, !queue.isEmpty {
            spawned.append(queue.removeFirst().kind)
            if let next = queue.first {
                timer += next.wait
            }
        }
        if queue.isEmpty {
            state = waveIndex == waves.count - 1 ? .finished : .waitingForClear
        } else {
            state = .spawning(timer: timer)
        }
        return spawned
    }

    private static func schedule(for wave: WaveConfig) -> [PendingSpawn] {
        wave.groups.flatMap { group in
            Array(repeating: PendingSpawn(kind: group.kind, wait: group.interval), count: max(0, group.count))
        }
    }
}

/// 船怎么选打捞目标。只做计算，不改状态。
public enum BoatAI {
    /// 船追上一个漂流物要多久；追不上返回 nil
    public static func interceptTime(boatX: Double, targetX: Double, boatSpeed: Double, flowSpeed: Double) -> Double? {
        let gap = targetX - boatX
        if gap <= 0 {
            // 目标在上游，迎面靠近
            return -gap / (boatSpeed + flowSpeed)
        }
        let closing = boatSpeed - flowSpeed
        guard closing > 0 else { return nil }
        return gap / closing
    }

    /// 漂流物还要多久漂出屏幕
    public static func timeToExit(x: Double, river: RiverConfig) -> Double {
        (river.exitX - x) / river.flowSpeed
    }

    public static func isReachable(boatX: Double, targetX: Double, boat: BoatConfig, river: RiverConfig) -> Bool {
        guard let time = interceptTime(boatX: boatX, targetX: targetX, boatSpeed: boat.speed, flowSpeed: river.flowSpeed) else {
            return false
        }
        return time < timeToExit(x: targetX, river: river)
    }

    /// 先排除追不上的，再挑最快会漂出屏幕的
    public static func chooseTarget(boatX: Double, carcasses: [Carcass], boat: BoatConfig, river: RiverConfig) -> EntityID? {
        carcasses
            .filter { $0.phase == .floating && isReachable(boatX: boatX, targetX: $0.position.x, boat: boat, river: river) }
            .min { timeToExit(x: $0.position.x, river: river) < timeToExit(x: $1.position.x, river: river) }?
            .id
    }
}

extension Battle {
    mutating func spawnDragons(dt: Double) {
        let world = balance.world
        for kindName in spawner.step(dt: dt, aliveDragons: dragons.count) {
            guard let kind = balance.dragons[kindName] else {
                preconditionFailure("未知的龙种类：\(kindName)")
            }
            let y = Double.random(in: world.spawnMinY...world.spawnMaxY, using: &rng)
            dragons.append(Dragon(id: makeID(), kind: kindName, position: Vec2(world.spawnX, y), hp: kind.maxHP))
        }
    }

    /// 龙飞向城墙，进入攻击距离后停下，按间隔远程喷火；不需要接触城墙
    mutating func moveDragons(dt: Double, events: inout [BattleEvent]) {
        for i in dragons.indices {
            let stats = dragonKind(dragons[i])
            switch dragons[i].phase {
            case .approaching:
                let stopX = balance.wall.frontX + stats.attackRange
                if dragons[i].position.x > stopX {
                    dragons[i].position.x = max(stopX, dragons[i].position.x - stats.speed * dt)
                }
                if dragons[i].position.x <= stopX {
                    dragons[i].phase = .attacking
                    dragons[i].attackCooldown = 0
                }
            case .attacking:
                dragons[i].attackCooldown -= dt
                if dragons[i].attackCooldown <= 0 {
                    dragons[i].attackCooldown += stats.attackInterval
                    wallHP = max(0, wallHP - stats.attackDamage)
                    events.append(.dragonAttacked(dragons[i].id))
                }
            }
        }
    }

    mutating func moveArrows(dt: Double, events: inout [BattleEvent]) {
        let gravity = balance.world.gravity
        var remaining: [Arrow] = []
        for var arrow in arrows {
            arrow.age += dt
            arrow.position = Trajectory.position(origin: arrow.origin, velocity: arrow.launchVelocity, gravity: gravity, time: arrow.age)
            arrow.velocity = Trajectory.velocity(launch: arrow.launchVelocity, gravity: gravity, time: arrow.age)
            let tip = arrow.position
            if let index = dragons.firstIndex(where: { $0.position.distance(to: tip) <= dragonKind($0).hitRadius }) {
                hitDragon(at: index, events: &events)
                continue
            }
            if isOutOfPlay(tip) { continue }
            remaining.append(arrow)
        }
        arrows = remaining
    }

    mutating func hitDragon(at index: Int, events: inout [BattleEvent]) {
        dragons[index].hp -= balance.bow.damage
        let dragon = dragons[index]
        guard dragon.hp <= 0 else {
            events.append(.dragonHit(dragon.id))
            return
        }
        // HP 归零即死亡；之后的下坠、漂流和打捞属于收集流程
        dragons.remove(at: index)
        dragonsKilled += 1
        carcasses.append(Carcass(id: dragon.id, kind: dragon.kind, value: dragonKind(dragon).value, position: dragon.position, phase: .falling))
        events.append(.dragonKilled(dragon.id))
    }

    /// 箭落进河里，或者飞出屏幕左右两侧，就移除
    func isOutOfPlay(_ point: Vec2) -> Bool {
        point.y < balance.river.surfaceY || point.x < -50 || point.x > balance.world.width + 50
    }
}

                if carcass.position.y <= river.floatY {
                    carcass.position.y = river.floatY
                    carcass.phase = .floating
                    events.append(.carcassLanded(carcass.id))
                }
            case .floating:
                carcass.position.x += river.flowSpeed * dt
                if carcass.position.x > river.exitX {
                    dragonsLost += 1
                    events.append(.carcassLost(carcass.id))
                    continue
                }
            }
            kept.append(carcass)
        }
        carcasses = kept
    }

    /// 船：选定目标后一直追，直到捞到、目标消失或追不上；没装满时如果出现新目标，返航途中也会掉头去捞；
    /// 装满或者没有目标时回码头卸货。M1 里打捞和卸货都是瞬间完成。
    mutating func moveBoat(dt: Double, events: inout [BattleEvent]) {
        let config = balance.boat
        if case .chasing(let id) = boat.mode, !isValidTarget(id) {
            boat.mode = .idle
        }
        if !isChasing {
            if boat.cargo.count < config.capacity,
               let id = BoatAI.chooseTarget(boatX: boat.x, carcasses: carcasses, boat: config, river: balance.river) {
                boat.mode = .chasing(id)
            } else if boat.cargo.isEmpty && boat.x == config.dockX {
                boat.mode = .idle
            } else {
                boat.mode = .returning
            }
        }

        switch boat.mode {
        case .idle:
            break
        case .chasing(let id):
            guard let index = carcasses.firstIndex(where: { $0.id == id }) else { break }
            boat.x = Self.approach(boat.x, toward: carcasses[index].position.x, maxStep: config.speed * dt)
            if abs(boat.x - carcasses[index].position.x) <= config.pickupRadius {
                let carcass = carcasses.remove(at: index)
                boat.cargo.append(carcass.value)
                boat.mode = .idle
                dragonsCollected += 1
                events.append(.carcassCollected(carcass.id))
            }
        case .returning:
            boat.x = Self.approach(boat.x, toward: config.dockX, maxStep: config.speed * dt)
            if boat.x == config.dockX {
                if !boat.cargo.isEmpty {
                    let earned = boat.cargo.reduce(0, +) * balance.economy.incomeMultiplier
                    income += earned
                    boat.cargo.removeAll()
                    events.append(.cargoUnloaded(income: earned))
                }
                boat.mode = .idle
            }
        }
    }

    /// 朝目标移动，每次最多移动 maxStep，不会越过目标
    static func approach(_ value: Double, toward target: Double, maxStep: Double) -> Double {
        let delta = target - value
        if abs(delta) <= maxStep { return target }
        return value + (delta > 0 ? maxStep : -maxStep)
    }

    private var isChasing: Bool {
        if case .chasing = boat.mode { return true }
        return false
    }

    private func isValidTarget(_ id: EntityID) -> Bool {
        guard let carcass = carcasses.first(where: { $0.id == id }), carcass.phase == .floating else { return false }
        return BoatAI.isReachable(boatX: boat.x, targetX: carcass.position.x, boat: balance.boat, river: balance.river)
    }
}

/// 把一组实体按 id 同步成节点：新出现的就建节点，已经消失的就删掉，其余的更新位置等属性。
@MainActor
final class EntityLayer<Item> {
    let root = SKNode()
    private var nodes: [EntityID: SKNode] = [:]
    private let id: (Item) -> EntityID
    private let make: @MainActor (Item) -> SKNode
    private let update: @MainActor (Item, SKNode) -> Void

    init(
        id: @escaping (Item) -> EntityID,
        make: @escaping @MainActor (Item) -> SKNode,
        update: @escaping @MainActor (Item, SKNode) -> Void
    ) {
        self.id = id
        self.make = make
        self.update = update
    }

    func node(for id: EntityID) -> SKNode? {
        nodes[id]
    }

    func sync(_ items: [Item]) {
        var seen = Set<EntityID>()
        for item in items {
            let key = id(item)
            seen.insert(key)
            let node: SKNode
            if let existing = nodes[key] {
                node = existing
            } else {
                node = make(item)
                root.addChild(node)
                nodes[key] = node
            }
            update(item, node)
        }
        for (key, node) in nodes where !seen.contains(key) {
            node.removeFromParent()
            nodes[key] = nil
        }
    }
}

import DragonCore
import SpriteKit

/// 一关的画面层：每帧推进 Battle，再把局内状态画出来。规则都在 DragonCore 里，这里只负责显示和触摸。
final class GameScene: SKScene {
    /// 一关结束（胜利收尾完成或失败）时调用一次
    var onFinish: (@MainActor (BattleSummary) -> Void)?

    private var battle: Battle
    private var lastUpdateTime: TimeInterval?
    private var reportedFinish = false

    private let wallNode: SKShapeNode
    private let archerNode: SKNode
    private let boatNode: SKNode
    private let hud: HUDNode
    private let aimPreview = AimPreviewNode()
    private let effectsLayer = SKNode()
    private let dragonLayer: EntityLayer<Dragon>
    private let arrowLayer: EntityLayer<Arrow>
    private let carcassLayer: EntityLayer<Carcass>

    init(battle: Battle) {
        self.battle = battle
        let balance = battle.balance
        let size = CGSize(width: balance.world.width, height: balance.world.height)
        let hitRadius: (String) -> Double = { balance.dragons[$0]?.hitRadius ?? 24 }

        wallNode = NodeFactory.wall(balance.wall)
        archerNode = NodeFactory.archer(balance.bow)
        boatNode = NodeFactory.boat()
        hud = HUDNode(sceneSize: size)
        dragonLayer = EntityLayer(
            id: \.id,
            make: { NodeFactory.dragon(hitRadius: hitRadius($0.kind)) },
            update: { dragon, node in node.position = CGPoint(dragon.position) }
        )
        arrowLayer = EntityLayer(
            id: \.id,
            make: { _ in NodeFactory.arrow() },
            update: { arrow, node in
                node.position = CGPoint(arrow.position)
                node.zRotation = atan2(arrow.velocity.y, arrow.velocity.x)
            }
        )
        carcassLayer = EntityLayer(
            id: \.id,
            make: { NodeFactory.carcass(hitRadius: hitRadius($0.kind)) },
            update: { carcass, node in node.position = CGPoint(carcass.position) }
        )
        super.init(size: size)
        scaleMode = .aspectFill
        backgroundColor = SKColor(red: 0.53, green: 0.81, blue: 0.98, alpha: 1)
    }

    required init?(coder aDecoder: NSCoder) {
        fatalError("GameScene 只能用代码创建")
    }

    override func didMove(to view: SKView) {
        guard children.isEmpty else { return }
        let balance = battle.balance
        addChild(NodeFactory.river(balance.river, width: balance.world.width))
        addChild(wallNode)
        addChild(NodeFactory.dock(balance.boat, river: balance.river))
        addChild(archerNode)
        addChild(carcassLayer.root)
        boatNode.position = CGPoint(x: balance.boat.dockX, y: balance.river.floatY)
        addChild(boatNode)
        addChild(dragonLayer.root)
        addChild(arrowLayer.root)
        addChild(aimPreview)
        addChild(effectsLayer)
        addChild(hud)
        render()
    }

    override func update(_ currentTime: TimeInterval) {
        // 限制单帧最长 1/30 秒，避免切后台回来时一帧跳太远
        let dt = lastUpdateTime.map { min(currentTime - $0, 1.0 / 30) } ?? 0
        lastUpdateTime = currentTime
        let events = battle.step(dt: dt)
        play(events)
        render()
        reportFinishIfNeeded()
    }

    // MARK: 触摸

    override func touchesBegan(_ touches: Set<UITouch>, with event: UIEvent?) {
        guard let touch = touches.first else { return }
        battle.beginAim(at: Vec2(touch.location(in: self)))
    }

    override func touchesMoved(_ touches: Set<UITouch>, with event: UIEvent?) {
        guard let touch = touches.first else { return }
        battle.moveAim(to: Vec2(touch.location(in: self)))
    }

    override func touchesEnded(_ touches: Set<UITouch>, with event: UIEvent?) {
        play(battle.endAim())
    }

    override func touchesCancelled(_ touches: Set<UITouch>, with event: UIEvent?) {
        battle.cancelAim()
    }

    // MARK: 画面

    private func render() {
        let balance = battle.balance
        dragonLayer.sync(battle.dragons)
        arrowLayer.sync(battle.arrows)
        carcassLayer.sync(battle.carcasses)
        boatNode.position.x = battle.boat.x
        archerNode.alpha = battle.bow.isReady ? 1 : 0.6
        aimPreview.update(bow: battle.bow, points: battle.aimPreview())
        hud.update(
            wallHP: battle.wallHP, maxHP: balance.wall.maxHP, income: battle.income,
            wave: battle.spawner.waveIndex + 1, waveCount: battle.spawner.waveCount
        )
    }

    /// 播放特效。要在 render 之前调用，这样被打死的龙节点还没被删掉，事件里的 id 还能找到对应节点。
    private func play(_ events: [BattleEvent]) {
        let balance = battle.balance
        for event in events {
            switch event {
            case .dragonHit(let id):
                if let node = dragonLayer.node(for: id) {
                    Effects.flash(node)
                }
            case .dragonAttacked(let id):
                guard let dragon = battle.dragons.first(where: { $0.id == id }) else { break }
                let target = CGPoint(x: balance.wall.frontX, y: min(dragon.position.y, balance.wall.topY - 10))
                Effects.fireball(from: CGPoint(dragon.position), to: target, in: effectsLayer)
                Effects.shake(wallNode, home: .zero)
            case .carcassLanded(let id):
                if let carcass = battle.carcasses.first(where: { $0.id == id }) {
                    Effects.splash(at: CGPoint(carcass.position), in: effectsLayer)
                }
            case .cargoUnloaded(let income):
                Effects.floatingText(
                    "+\(Int(income))",
                    at: CGPoint(x: balance.boat.dockX, y: balance.river.surfaceY + 30),
                    in: effectsLayer
                )
            case .arrowFired, .dragonKilled, .carcassLost, .carcassCollected, .outcomeChanged:
                break
            }
        }
    }

    private func reportFinishIfNeeded() {
        guard !reportedFinish, battle.outcome == .won || battle.outcome == .lost else { return }
        reportedFinish = true
        onFinish?(battle.summary)
    }
}

```swift
import DragonCore
import Observation

/// 持有当前这一局的场景。restart 会换一个新场景，等于重置局内状态。
@MainActor
@Observable
final class GameSession {
    private(set) var scene: GameScene
    private(set) var summary: BattleSummary?

    init() {
        scene = GameScene.makeLevel1()
        attachFinishHandler()
    }

    func restart() {
        summary = nil
        scene = GameScene.makeLevel1()
        attachFinishHandler()
    }

    private func attachFinishHandler() {
        scene.onFinish = { [weak self] summary in
            self?.summary = summary
        }
    }
}
```
